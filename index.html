function doPost(e) { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (data.action === "GRADE_STUDENT") { // 1. Guardar en Google Sheets sheet.appendRow([ new Date(), data.email, "Práctica " + data.practice, data.finalGrade, "EVALUADO" ]); // 2. Transfere la nota a Google Classroom try { updateClassroomGrade(data.courseId, data.courseWorkId, data.email, data.finalGrade); } catch (err) { Logger.log("Error en Classroom: " + err.toString()); } return ContentService.createTextOutput(JSON.stringify({ status: "SUCCESS" })) .setMimeType(ContentService.MimeType.JSON); } } function updateClassroomGrade(courseId, courseWorkId, studentEmail, grade) { // Buscar entrega del alumno mediante la API de Classroom var submissions = Classroom.Courses.CourseWork.StudentSubmissions.list(courseId, courseWorkId, { userId: studentEmail }).studentSubmissions; if (submissions && submissions.length > 0) { var submission = submissions[0]; var studentSubmissionId = submission.id; // Actualizar la nota asignada var resource = { assignedGrade: parseFloat(grade), draftGrade: parseFloat(grade) }; Classroom.Courses.CourseWork.StudentSubmissions.patch( resource, courseId, courseWorkId, studentSubmissionId, { updateMask: "assignedGrade,draftGrade" } ); } }
